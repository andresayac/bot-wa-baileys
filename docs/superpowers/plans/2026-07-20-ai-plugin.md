# @bot-wa/ai Plugin Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Convertir el repo a monorepo pnpm y añadir el paquete `@bot-wa/ai`: un plugin de IA multi-proveedor (Anthropic/OpenAI/Ollama) con context manager por chat y autorespuesta a mensajes no-manejados.

**Architecture:** Fase 0 mueve el core actual a `packages/core` (sin cambios de código). Luego `packages/ai` define una interfaz `AIProvider` con 3 implementaciones (SDKs cargados lazy como deps opcionales), un `ContextManager` respaldado por el `IStore` del core, un `AIClient` que orquesta proveedor+contexto, y `aiPlugin()` que engancha un catch-all e inyecta `ctx.tools.ai`. Spec: `docs/superpowers/specs/2026-07-20-ai-plugin-design.md`.

**Tech Stack:** TypeScript ESM (NodeNext, Node ≥20), pnpm workspaces, vitest, `@anthropic-ai/sdk`/`openai` (optional), Ollama vía fetch.

## Global Constraints

- ESM-only, Node ≥ 20; imports relativos con extensión `.js`; `"type": "module"`.
- Cero `console.*` en `src/` (usar el `logger` inyectado); cero `any` en tipos exportados públicos.
- SDKs LLM (`@anthropic-ai/sdk`, `openai`) son `optionalDependencies` + `peerDependenciesMeta.optional`; se cargan con `await import()`; si faltan → error accionable, nunca crash.
- Modelo por defecto solo para Anthropic: `claude-opus-4-8`. OpenAI y Ollama requieren `model` en config (sin default).
- Anthropic: `messages.create/.stream` con `thinking: { type: 'adaptive' }`; sin `temperature`/`top_p`/`top_k`; `max_tokens` default 1024; manejar `stop_reason === 'refusal'`.
- Commits: Conventional Commits, cuerpo termina con `Co-Authored-By: Claude Fable 5 <noreply@anthropic.com>`.
- Tests: vitest. Comando raíz `pnpm -r test`; por paquete `pnpm --filter @bot-wa/ai test`.

// packages/ai/src/MockAIProvider.ts
import type { AIProvider, GenerateInput, GenerateResult } from './types.js'

export interface MockOptions { reply?: string | ((input: GenerateInput) => string); model?: string }

export class MockAIProvider implements AIProvider {
  readonly name = 'mock'
  readonly calls: GenerateInput[] = []
  constructor(private opts: MockOptions = {}) {}

  private textFor(input: GenerateInput): string {
    const r = this.opts.reply ?? 'mock reply'
    return typeof r === 'function' ? r(input) : r
  }
  async generate(input: GenerateInput): Promise<GenerateResult> {
    this.calls.push(input)
    return { text: this.textFor(input), model: this.opts.model ?? 'mock', usage: { input: 0, output: 0 } }
  }
  async *stream(input: GenerateInput): AsyncIterable<string> {
    this.calls.push(input)
    for (const ch of this.textFor(input)) yield ch
  }
}

/** Chat-key con la fórmula del core (scope participante por defecto). */
export function chatKeyOf(ctx: { from: string; sender: string; isGroup: boolean }): string {
  return ctx.isGroup ? `${ctx.from}:${ctx.sender}` : ctx.from

```ts
// packages/ai/src/ContextManager.ts
import type { AIMessage } from '@bot-wa/bot-wa-baileys'

export interface ConvStore {
  getConversation(k: string): Promise<AIMessage[] | undefined>
  setConversation(k: string, m: AIMessage[]): Promise<void>
  clearConversation(k: string): Promise<void>
}

export class ContextManager {
  private max: number
  constructor(private store: ConvStore, opts: { maxMessages?: number } = {}) {
    this.max = opts.maxMessages ?? 20
  }
  async history(key: string): Promise<AIMessage[]> {
    return (await this.store.getConversation(key)) ?? []
  }
  async append(key: string, msg: AIMessage): Promise<void> {
    const h = await this.history(key)
    h.push(msg)
    const trimmed = h.length > this.max ? h.slice(h.length - this.max) : h
    await this.store.setConversation(key, trimmed)
  }
  async clear(key: string): Promise<void> {
    await this.store.clearConversation(key)
  }
}
```

// packages/ai/src/AIClient.ts
import type { AIMessage } from '@bot-wa/bot-wa-baileys'
import type { AIProvider, GenerateInput, GenerateResult, Effort } from './types.js'
import type { ContextManager } from './ContextManager.js'

export interface AIClientOptions {
  persona?: string
  model?: string
  maxTokens?: number
  effort?: Effort
  refusalMessage?: string
}

export class AIClient {
  constructor(
    private provider: AIProvider,
    private ctx: ContextManager,
    private opts: AIClientOptions = {},
  ) {}

  private base(messages: AIMessage[]): GenerateInput {
    return {
      system: this.opts.persona,
      messages,
      model: this.opts.model,
      maxTokens: this.opts.maxTokens,
      effort: this.opts.effort,
    }
  }

  async ask(prompt: string): Promise<string> {
    const r = await this.provider.generate(this.base([{ role: 'user', content: prompt }]))
    return r.stopReason === 'refusal' ? (this.opts.refusalMessage ?? 'No puedo ayudar con eso.') : r.text
  }

  async generate(input: Partial<GenerateInput> & { messages: AIMessage[] }): Promise<GenerateResult> {
    return this.provider.generate({ ...this.base(input.messages), ...input })
  }

  stream(input: Partial<GenerateInput> & { messages: AIMessage[] }): AsyncIterable<string> {
    return this.provider.stream({ ...this.base(input.messages), ...input })
  }

  async replyInChat(chatKey: string, userText: string): Promise<string> {
    const history = await this.ctx.history(chatKey)
    const messages: AIMessage[] = [...history, { role: 'user', content: userText }]
    const r = await this.provider.generate(this.base(messages))
    if (r.stopReason === 'refusal') {
      return this.opts.refusalMessage ?? 'No puedo ayudar con eso.'
    }
    await this.ctx.append(chatKey, { role: 'user', content: userText })
    await this.ctx.append(chatKey, { role: 'assistant', content: r.text })
    return r.text
  }
}

// packages/ai/src/providers/anthropic.ts
import type { AIProvider, GenerateInput, GenerateResult, AIConfig } from '../types.js'

export interface AnthropicLike {
  messages: {
    create(args: Record<string, unknown>): Promise<{
      content: Array<{ type: string; text?: string }>
      stop_reason?: string
      model: string
      usage?: { input_tokens: number; output_tokens: number }
    }>
    stream?(args: Record<string, unknown>): AsyncIterable<any> & { getFinalMessage?: () => Promise<unknown> }
  }
}

const DEFAULT_MODEL = 'claude-opus-4-8'

export class AnthropicProvider implements AIProvider {
  readonly name = 'anthropic'
  constructor(private client: AnthropicLike, private cfg: AIConfig) {}

  private buildArgs(input: GenerateInput): Record<string, unknown> {
    const args: Record<string, unknown> = {
      model: input.model ?? this.cfg.model ?? DEFAULT_MODEL,
      max_tokens: input.maxTokens ?? this.cfg.maxTokens ?? 1024,
      messages: input.messages,
      thinking: { type: 'adaptive' },
    }
    if (input.system) args.system = input.system
    const effort = input.effort ?? this.cfg.effort
    if (effort) args.output_config = { effort }
    return args
  }

  async generate(input: GenerateInput): Promise<GenerateResult> {
    const res = await this.client.messages.create(this.buildArgs(input))
    const text = res.content.filter((b) => b.type === 'text').map((b) => b.text ?? '').join('')
    return {
      text,
      model: res.model,
      stopReason: res.stop_reason,
      usage: res.usage ? { input: res.usage.input_tokens, output: res.usage.output_tokens } : undefined,
    }
  }

  async *stream(input: GenerateInput): AsyncIterable<string> {
    if (!this.client.messages.stream) {
      const r = await this.generate(input)
      if (r.text) yield r.text
      return
    }
    const s = this.client.messages.stream(this.buildArgs(input))
    for await (const event of s as AsyncIterable<any>) {
      if (event?.type === 'content_block_delta' && event.delta?.type === 'text_delta') {
        yield event.delta.text as string
      }
    }
  }
}

import type { AIProvider, GenerateInput, GenerateResult, AIConfig } from '../types.js'

export interface OpenAILike {
  chat: {
    completions: {
      create(args: Record<string, unknown>): Promise<{
        choices: Array<{ message?: { content?: string }; delta?: { content?: string }; finish_reason?: string }>
        model: string
        usage?: { prompt_tokens: number; completion_tokens: number }
      } | AsyncIterable<any>>
    }
  }
}

export class OpenAIProvider implements AIProvider {
  readonly name = 'openai'
  constructor(private client: OpenAILike, private cfg: AIConfig) {}

  private messagesFor(input: GenerateInput) {
    const msgs = input.system ? [{ role: 'system', content: input.system }, ...input.messages] : [...input.messages]
    return msgs
  }
  private model(input: GenerateInput): string {
    const m = input.model ?? this.cfg.model
    if (!m) throw new Error('OpenAIProvider requiere `model` en la config (sin default).')
    return m
  }

  async generate(input: GenerateInput): Promise<GenerateResult> {
    const res = (await this.client.chat.completions.create({
      model: this.model(input),
      messages: this.messagesFor(input),
      max_tokens: input.maxTokens ?? this.cfg.maxTokens ?? 1024,
    })) as { choices: Array<{ message?: { content?: string }; finish_reason?: string }>; model: string; usage?: { prompt_tokens: number; completion_tokens: number } }
    return {
      text: res.choices[0]?.message?.content ?? '',
      model: res.model,
      stopReason: res.choices[0]?.finish_reason,
      usage: res.usage ? { input: res.usage.prompt_tokens, output: res.usage.completion_tokens } : undefined,
    }
  }

  async *stream(input: GenerateInput): AsyncIterable<string> {
    const s = (await this.client.chat.completions.create({
      model: this.model(input),
      messages: this.messagesFor(input),
      max_tokens: input.maxTokens ?? this.cfg.maxTokens ?? 1024,
      stream: true,
    })) as AsyncIterable<{ choices: Array<{ delta?: { content?: string } }> }>
    for await (const chunk of s) {
      const d = chunk.choices[0]?.delta?.content
      if (d) yield d
    }
  }
}

// packages/ai/src/providers/ollama.ts
import type { AIProvider, GenerateInput, GenerateResult, AIConfig } from '../types.js'

export class OllamaProvider implements AIProvider {
  readonly name = 'ollama'
  private base: string
  constructor(private cfg: AIConfig, private fetchImpl: typeof fetch = fetch) {
    this.base = (cfg.baseUrl ?? 'http://localhost:11434').replace(/\/$/, '')
  }
  private model(input: GenerateInput): string {
    const m = input.model ?? this.cfg.model
    if (!m) throw new Error('OllamaProvider requiere `model` en la config (sin default).')
    return m
  }
  private messagesFor(input: GenerateInput) {
    return input.system ? [{ role: 'system', content: input.system }, ...input.messages] : [...input.messages]
  }

  async generate(input: GenerateInput): Promise<GenerateResult> {
    const res = await this.fetchImpl(`${this.base}/api/chat`, {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({ model: this.model(input), messages: this.messagesFor(input), stream: false }),
      signal: input.signal,
    })
    if (!res.ok) throw new Error(`Ollama respondió ${res.status}`)
    const data = (await res.json()) as { message?: { content?: string }; model?: string }
    return { text: data.message?.content ?? '', model: data.model ?? this.model(input) }
  }

  async *stream(input: GenerateInput): AsyncIterable<string> {
    const res = await this.fetchImpl(`${this.base}/api/chat`, {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({ model: this.model(input), messages: this.messagesFor(input), stream: true }),
      signal: input.signal,
    })
    if (!res.ok || !res.body) throw new Error(`Ollama respondió ${res.status}`)
    const reader = (res.body as any).getReader()
    const decoder = new TextDecoder()
    let buf = ''
    for (;;) {
      const { done, value } = await reader.read()
      if (done) break
      buf += decoder.decode(value, { stream: true })
      const lines = buf.split('\n')
      buf = lines.pop() ?? ''
      for (const line of lines) {
        if (!line.trim()) continue
        const obj = JSON.parse(line) as { message?: { content?: string } }
        if (obj.message?.content) yield obj.message.content
      }
    }
  }
}

import type { AIProvider, AIConfig } from '../types.js'

export async function resolveProvider(cfg: AIConfig): Promise<AIProvider> {
  switch (cfg.provider ?? 'anthropic') {
    case 'anthropic': {
      const mod = await loadModule<{ default: new (o: { apiKey?: string }) => any }>('@anthropic-ai/sdk', 'pnpm add @anthropic-ai/sdk')
      return new AnthropicProvider(new mod.default({ apiKey: cfg.apiKey }), cfg)
    }
    case 'openai': {

      return new OpenAIProvider(new mod.default({ apiKey: cfg.apiKey, baseURL: cfg.baseUrl }), cfg)
    }
    case 'ollama':
      return new OllamaProvider(cfg)
    default:
      throw new Error(`Proveedor desconocido: ${cfg.provider}`)
  }
}

// packages/ai/src/plugin.ts
import { definePlugin, type PluginContext } from '@bot-wa/bot-wa-baileys'
import type { AIConfig, AIProvider } from './types.js'
import { ContextManager } from './ContextManager.js'
import { AIClient } from './AIClient.js'
import { resolveProvider } from './providers/resolve.js'
import { chatKeyOf } from './chatKey.js'

type PluginConfig = AIConfig & { __provider?: AIProvider }

export function aiPlugin(config: PluginConfig = {}) {
  return definePlugin<PluginConfig>({
    name: 'ai',
    version: '0.1.0',
    description: 'Multi-provider AI autoresponder + ctx.tools.ai',
    async install(ctx: PluginContext, cfg: PluginConfig = config) {
      const bot: any = ctx.bot
      const provider = cfg.__provider ?? (await resolveProvider(cfg))
      const ctxMgr = new ContextManager(bot.getStore(), { maxMessages: cfg.context?.maxMessages })
      const client = new AIClient(provider, ctxMgr, {
        persona: cfg.persona,
        model: cfg.model,
        maxTokens: cfg.maxTokens,
        effort: cfg.effort,
        refusalMessage: cfg.refusalMessage,
      })
      bot.provide('ai', client)

      const errorMessage = cfg.errorMessage ?? '⚠️ Ahora no puedo responder, intenta luego.'

      if (cfg.autoReply !== false) {
        bot.onMessage(async (mctx: any) => {
          if (mctx.type !== 'text' || !mctx.body) return
          const key = chatKeyOf(mctx)
          try {
            if (cfg.humanizeTyping !== false) await mctx.typing()
            const text = await client.replyInChat(key, mctx.body)
            await mctx.reply(text)
          } catch (err) {
            ctx.logger.warn({ err }, 'ai autoreply failed')
            await mctx.reply(errorMessage)
          }
        })
      }

      if (cfg.resetCommand !== false) {
        bot.command('reset', async (mctx: any) => {
          await ctxMgr.clear(chatKeyOf(mctx))
          bot.abortConversation(mctx)
          await mctx.reply('✅ Conversación reiniciada.')
        })
      }
    },
  })
}
